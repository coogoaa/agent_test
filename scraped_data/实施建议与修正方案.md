// 修改 solar-calculator.js
const STRATEGY_CONFIG = {
    A: { name: "高端型", battery_ratio: 1.5 },
    B: { name: "平衡型", battery_ratio: 1.0 },
    C: { name: "经济型", battery_ratio: 0.4 },
    D: { name: "纯光伏型", battery_ratio: 0 }  // 新增：无电池方案

// 修改电池定容函数
function step2_sizeBattery(loadAnalysis, strategyType) {
    const nightlyLoad = loadAnalysis.nightlyConsumptionKwh;
    const ratio = STRATEGY_CONFIG[strategyType].battery_ratio;
    
    // 如果是纯光伏方案，直接返回0
    if (ratio === 0) {
        return {
            targetCapacity: 0,
            selectedBatteryKwh: 0,
            ratio: 0
        };
    }
    
    const targetCapacity = nightlyLoad * ratio;
    const standardBatteries = [5.0, 9.6, 13.5, 19.2];
    let selectedBatteryKwh = 5.0;
    
    for (let bat of standardBatteries) {
        if (bat >= targetCapacity * 0.9) {
            selectedBatteryKwh = bat;
            break;
        }
    }
    
    if (targetCapacity > standardBatteries[standardBatteries.length - 1]) {
        selectedBatteryKwh = standardBatteries[standardBatteries.length - 1];
    }
    
    return {
        targetCapacity,
        selectedBatteryKwh,
        ratio
    };
}

// 修改逆变器选型（无电池时使用普通逆变器）
function step3_selectInverter(batteryKwh, peakLoadKw) {
    // 如果没有电池，使用普通逆变器（不需要混合功能）
    if (batteryKwh === 0) {
        const standardInverters = [3.0, 5.0, 6.0, 8.0, 10.0];
        const targetInverterKw = peakLoadKw * 0.7;
        
        let selectedInverterKw = 5.0;
        for (let inv of standardInverters) {
            if (inv >= targetInverterKw) {
                selectedInverterKw = inv;
                break;
            }
        }
        
        return {
            minInverterKwByBattery: 0,
            targetInverterKw,
            selectedInverterKw,
            inverterType: 'standard'  // 标记为普通逆变器
        };
    }
    
    // 有电池时使用混合逆变器（原逻辑）
    const minInverterKwByBattery = batteryKwh * 0.5;
    const standardInverters = [5.0, 6.0, 8.0, 10.0];
    const targetInverterKw = Math.max(minInverterKwByBattery, peakLoadKw * 0.7);
    
    let selectedInverterKw = 5.0;
    for (let inv of standardInverters) {
        if (inv >= targetInverterKw) {
            selectedInverterKw = inv;
            break;
        }
    }
    
    if (selectedInverterKw > 10.0) {
        selectedInverterKw = 10.0;
    }
    
    return {
        minInverterKwByBattery,
        targetInverterKw,
        selectedInverterKw,
        inverterType: 'hybrid'  // 标记为混合逆变器
    };
}

/**
 * 将计算的系统容量调整到最接近的市场常见值
 */
function adjustToMarketSize(calculatedKw) {
    const commonSizes = MARKET_DATA.common_sizes;
    
    // 找到最接近的标准容量
    let closest = commonSizes[0];
    let minDiff = Math.abs(calculatedKw - closest);
    
    for (let size of commonSizes) {
        const diff = Math.abs(calculatedKw - size);
        if (diff < minDiff) {
            minDiff = diff;
            closest = size;
        }
    }
    
    return {
        original: calculatedKw,
        adjusted: closest,
        reason: minDiff > 1.0 ? `调整到市场常见容量 ${closest}kW` : '符合市场标准'
    };
}

/**
 * 验证价格是否在合理范围内
 */
function validatePrice(systemKw, batteryKwh, calculatedCost, state) {
    // 找到最接近的容量档位
    const sizeKey = findClosestSize(systemKw, Object.keys(MARKET_DATA.price_ranges_no_battery));
    const baseRange = MARKET_DATA.price_ranges_no_battery[sizeKey];
    
    if (!baseRange) {
        return { valid: true, warning: '无市场数据参考' };
    }
    
    // 计算预期价格范围
    let expectedMin = baseRange.min;
    let expectedMax = baseRange.max;
    
    // 加上电池成本
    if (batteryKwh > 0) {
        const batteryKey = findClosestSize(batteryKwh, Object.keys(MARKET_DATA.battery_price_increment));
        const batteryCost = MARKET_DATA.battery_price_increment[batteryKey] || 10000;
        expectedMin += batteryCost * 0.8;
        expectedMax += batteryCost * 1.2;
    }
    
    // 应用州系数
    const stateMultiplier = MARKET_DATA.state_price_multiplier[state] || 1.0;
    expectedMin *= stateMultiplier;
    expectedMax *= stateMultiplier;
    
    // 验证
    if (calculatedCost < expectedMin * 0.8) {
        return {
            valid: false,
            warning: `价格过低（${calculatedCost.toFixed(0)}），市场范围 $${expectedMin.toFixed(0)}-$${expectedMax.toFixed(0)}`,
            expectedRange: { min: expectedMin, max: expectedMax }
        };
    }
    
    if (calculatedCost > expectedMax * 1.3) {
        return {
            valid: false,
            warning: `价格过高（${calculatedCost.toFixed(0)}），市场范围 $${expectedMin.toFixed(0)}-$${expectedMax.toFixed(0)}`,
            expectedRange: { min: expectedMin, max: expectedMax }
        };
    }
    
    return {
        valid: true,
        message: `价格合理，在市场范围内 ($${expectedMin.toFixed(0)}-$${expectedMax.toFixed(0)})`,
        expectedRange: { min: expectedMin, max: expectedMax }
    };
}

/**
 * 提供电池配置建议
 */
function getBatteryRecommendation(systemKw, state, userBudget) {
    const sizeKey = findClosestSize(systemKw, Object.keys(MARKET_DATA.battery_adoption_by_size));
    const marketRate = MARKET_DATA.battery_adoption_by_size[sizeKey] || 0.12;
    
    // 根据市场配置率给出建议
    if (marketRate < 0.1) {
        return {
            recommend: false,
            reason: `该容量段(${systemKw}kW)仅${(marketRate*100).toFixed(1)}%的用户选择电池，建议优先考虑纯光伏方案`,
            marketRate: marketRate
        };
    } else if (marketRate > 0.15) {
        return {
            recommend: true,
            reason: `该容量段(${systemKw}kW)有${(marketRate*100).toFixed(1)}%的用户选择电池，建议考虑电池方案`,
            marketRate: marketRate
        };
    } else {
        return {
            recommend: 'optional',
            reason: `该容量段(${systemKw}kW)有${(marketRate*100).toFixed(1)}%的用户选择电池，可根据预算选择`,
            marketRate: marketRate
        };
    }
}

/**
 * 辅助函数：找到最接近的尺寸
 */
function findClosestSize(value, sizeArray) {
    const sizes = sizeArray.map(s => parseFloat(s)).sort((a, b) => a - b);
    let closest = sizes[0];
    let minDiff = Math.abs(value - closest);
    
    for (let size of sizes) {
        const diff = Math.abs(value - size);
        if (diff < minDiff) {
            minDiff = diff;
            closest = size;
        }
    }
    
    return closest;
}

// 新增价格计算函数
function calculateSystemCost(systemKw, batteryKwh, inverterKw, state, strategy) {
    // 基础光伏系统成本（每kW）
    const basePricePerKw = {
        A: 1200,  // 高端：使用premium品牌
        B: 900,   // 平衡：使用balanced品牌
        C: 700,   // 经济：使用budget品牌
        D: 650    // 纯光伏：最经济
    }[strategy] || 900;
    
    // 光伏系统成本
    let pvCost = systemKw * basePricePerKw;
    
    // 逆变器成本
    const inverterCostPerKw = batteryKwh > 0 ? 280 : 200;  // 混合逆变器更贵
    let inverterCost = inverterKw * inverterCostPerKw;
    
    // 电池成本
    let batteryCost = 0;
    if (batteryKwh > 0) {
        const batteryPricePerKwh = {
            A: 1000,  // 高端电池（Tesla）
            B: 865,   // 平衡（Sungrow/BYD）
            C: 750    // 经济（Alpha-ESS）
        }[strategy] || 865;
        
        batteryCost = batteryKwh * batteryPricePerKwh;
    }
    
    // 安装和其他成本
    const installationCost = 1500 + (systemKw * 100);
    
    // 小计
    let subtotal = pvCost + inverterCost + batteryCost + installationCost;
    
    // 应用州系数
    const stateMultiplier = MARKET_DATA.state_price_multiplier[state] || 1.0;
    subtotal *= stateMultiplier;
    
    // GST
    const gst = subtotal * 0.1;
    const totalBeforeRebate = subtotal + gst;
    
    // 计算补贴
    const rebates = calculateRebates(systemKw, batteryKwh, state);
    const totalAfterRebate = totalBeforeRebate - rebates.total;
    
    return {
        breakdown: {
            pv_system: pvCost,
            inverter: inverterCost,
            battery: batteryCost,
            installation: installationCost,
            subtotal: subtotal,
            gst: gst,
            total_before_rebate: totalBeforeRebate
        },
        rebates: rebates,
        total_after_rebate: totalAfterRebate,
        
        // 🆕 市场对比
        market_comparison: compareWithMarket(systemKw, batteryKwh, totalAfterRebate, state)
    };
}

function compareWithMarket(systemKw, batteryKwh, calculatedCost, state) {
    const validation = validatePrice(systemKw, batteryKwh, calculatedCost, state);
    
    if (!validation.expectedRange) {
        return { message: '无市场数据参考' };
    }
    
    const marketAvg = (validation.expectedRange.min + validation.expectedRange.max) / 2;
    const diff = calculatedCost - marketAvg;
    const diffPct = (diff / marketAvg * 100).toFixed(1);
    
    if (Math.abs(diff) < marketAvg * 0.1) {
        return {
            status: 'good',
            message: `价格与市场平均值接近 (${diffPct > 0 ? '+' : ''}${diffPct}%)`,
            market_avg: marketAvg
        };
    } else if (diff < 0) {
        return {
            status: 'excellent',
            message: `价格低于市场平均 ${Math.abs(diffPct)}%，性价比优秀`,
            market_avg: marketAvg,
            savings: Math.abs(diff)
        };
    } else {
        return {
            status: 'high',
            message: `价格高于市场平均 ${diffPct}%`,
            market_avg: marketAvg,
            premium: diff
        };
    }
}
