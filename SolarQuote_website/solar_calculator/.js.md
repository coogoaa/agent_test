/**
 * Centralized JS for electricity plan selection across multiple calculators.
 */

(function ($) {
  'use strict';

  $(document).ready(function () {
    // Create a global or window-scoped object so other scripts can access it if needed.
    const ElectricityPlanSelection = {
      // Data / state
      plansDiscounts: {},
      retailersList: [],
      plansList: [],
      loadingPlanData: false,

      /**
       * Initialize the plan selection script
       */
      init: function () {
        // Bind events and do any initial setup
        this.bindEvents();

        // If there's a pre-filled postcode, automatically load the retailers
        if ($('#postcode').val()) {
          $('.create-plan-dropdown').removeClass('disabled');
          this.updateRetailersList();
        }

        // If the page already has an array of energyPlans (global variable), load them
        if (
          typeof window.energyPlans !== 'undefined' &&
          window.energyPlans.length > 0
        ) {
          this.loadEnergyPlans(window.energyPlans, true);
        }

        // Init preact popup for plan creation
        this.initCreatePlanPopup();
      },

      /**
       * All event bindings in one place
       */
      bindEvents: function () {
        // When user changes the postcode, fetch retailer list for that postcode
        $('#postcode').on('keyup change', (e) => {
          // Note: we might want to wait until user has 3 digits
          if ($(e.target).val().length >= 3) {
            sCalc.getPostcodeInfo($(e.target), window.calcs.stateDefaults, {
              callback: (oldState, newState) => {
                $('.create-plan-dropdown').removeClass('disabled');
                // If the state changed, you might want to reset or clear the plan
                if (oldState !== newState) {
                  ElectricityPlanSelection.resetPlan();
                }
                ElectricityPlanSelection.updateRetailersList();
              },
            });
          }
        });

        // .create-plan-dropdown: open the create-plan popup if postcode is not empty
        $('.create-plan-dropdown').on('click', function () {
          if ($('#postcode').val() !== '') {
            $('a[data-popup="create-plan"]')[0].click();
          }
        });

        // Prevent direct editing of .readonly input
        $('input.readonly').on('focus', function () {
          document.activeElement.blur();
        });
      },

      /**
       * Reset the plan selection to "empty"
       */
      resetPlan: function () {
        $("input[name='custom-plan-display-name']").val('');
        $('#plan-changes').val('');
        $('#custom-plan').val('');
        const retailerSelect = document.querySelector(
          '[name="retailer-selection-value"]'
        );
        if (retailerSelect) {
          retailerSelect.dispatchEvent(new Event('change'));
        }
      },

      /**
       * Load the plan array (either from the server or from a global JS var)
       * and store them for later use.
       *
       * @param {Array} plans
       * @param {boolean} firstLoad
       */
      loadEnergyPlans: function (plans, firstLoad = false) {
        // Sort by name
        plans.sort((a, b) =>
          a.name.toLowerCase().localeCompare(b.name.toLowerCase())
        );

        this.plansList = plans;

        for (let i = 0; i < plans.length; i++) {
          this.plansDiscounts[plans[i].offerId] =
            plans[i].availablePercentageDiscounts;
        }

        if (($('#custom-plan').val() ?? '') !== '') {
          setTimeout(() => {
            this.updateRetailersList();
          }, 200);
        }

        if (typeof window.resetSqDropdowns !== 'undefined') {
          resetSqDropdowns();
        }
      },

      /**
       * Make an AJAX call to fetch the retailer list for the given postcode
       */
      updateRetailersList: function () {
        const postcode = $('#postcode').val();
        if (!postcode) return;

        $.post({
          url: '/battery-storage/calculator/ajaxSearchRetailers/',
          data: { postcode: postcode },
          success: (result) => {
            this.retailersList = result.retailers || [];

            const retailerSelectionEl = $('#retailer-selection')[0];
            if (retailerSelectionEl) {
              retailerSelectionEl.setAttribute('placeholder', 'Loading...');
            }

            const retailerDropdown = $('.sqDropdown.retailer-selection')[0];
            if (retailerDropdown) {
              retailerDropdown.clearOptions();
              retailerDropdown.setDropdownValue(null, null);

              this.retailersList.forEach((retailer) => {
                retailerDropdown.addOption(
                  `<p class="retailer-name">${retailer}</p>`,
                  retailer
                );
              });
            }

            if (retailerSelectionEl) {
              retailerSelectionEl.setAttribute(
                'placeholder',
                'Select a retailer'
              );
            }

            if (typeof resetSqDropdowns !== 'undefined') {
              resetSqDropdowns();
            }

            // If user has a previously selected retailer in localStorage, re-select it
            const prevSel = selectedPlan?.providerName ?? localStorage.getItem('retailer-create-plan');
            if (prevSel && this.retailersList.includes(prevSel)) {
              // If it's a custom plan, maybe we do not trigger the event
              const hasCustomPlan = $('#custom-plan').val() !== '';
              const triggerChange = !hasCustomPlan;
              retailerDropdown.setDropdownValue(
                prevSel,
                prevSel,
                triggerChange
              );
            }
          },
          error: (err) => {
            console.log('Error fetching retailer list:', err);
          },
        });
      },

      /**
       * Initialize the Preact-based "Create Plan" popup
       */
      initCreatePlanPopup: function () {
        const planCreationContainer = document.getElementById('planCreationContainer');
        if (!planCreationContainer || typeof window.renderCreateEnergyPlanPopup === 'undefined') {
          return; // not on this page or not using the popup
        }
        window.renderCreateEnergyPlanPopup({
          container: planCreationContainer,
          initialEnergyPlan: selectedPlan,
          initialRatesStructure: ratesStructure,
          ajaxSearchDefaultPlan: ({
            retailer,
            planType,
            multipleTariffPeriods,
            controlledLoad,
          }) => {
            return fetch('/battery-storage/calculator/ajaxSearchDefaultPlan/', {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify({
                planType,
                multipleTariffPeriods,
                controlledLoad,
                retailer,
                postcode: $('#postcode').val(),
                state: $('#postcode-state').val(),
              }),
            });
          },
          onPlanSaved: (plan) => {
            function serializeForm(formEl) {
              const formData = new FormData(formEl);
              const result = {};
            
              for (let [name, value] of formData.entries()) {
                if (!name) continue;
                const keys = name.match(/[^\[\]]+/g); // e.g., solararray[1][tilt] => ['solararray', '1', 'tilt']
                let ref = result;
            
                keys.forEach((key, index) => {
                  if (index === keys.length - 1) {
                    // Last key: assign value
                    if (ref[key] !== undefined) {
                      if (!Array.isArray(ref[key])) {
                        ref[key] = [ref[key]];
                      }
                      ref[key].push(value);
                    } else {
                      ref[key] = value;
                    }
                  } else {
                    if (ref[key] === undefined) ref[key] = {};
                    ref = ref[key];
                  }
                });
              }
              return result;
            }

            // Save the plan JSON into the hidden form field
            $('#custom-plan').val(JSON.stringify(plan));

            const retailerName = plan?.planData?.retailerName || '';
              const tariffType = {
                SR: 'Single Rate',
                TOU: 'Time of Use',
              }[plan?.planData?.tariffType] || 'Custom';

            const customPlanDisplayName = `${retailerName} ${tariffType} Plan`
            
            const form = document.querySelector(".form-calculator");
            const dataObj = serializeForm(form);
            dataObj['custom-plan-display-name'] = customPlanDisplayName;
            
            const json = JSON.stringify(dataObj);
            const encoded = encodeURIComponent(btoa(json)); // base64 + URL encode

            const url = new URL(window.location.href);
            url.searchParams.set('data', encoded); // Add or replace `data` parameter
            url.hash = '!create-plan'; // Set or update the hash

            let targetUrl = url.toString();

            // Only navigate if something actually changed
            if (targetUrl === window.location.href) {
              url.hash = 'postcode-input';
              targetUrl = url.toString();
              window.location.href = targetUrl;
              $.closePopup();
            } else {
              url.hash = 'postcode-input';
              targetUrl = url.toString();
              window.location.href = targetUrl;
            }
          },
        });
      },
    };

    // On DOM Ready, initialize
    $(document).ready(function () {
      ElectricityPlanSelection.init();
    });

    // Expose globally in case we want to call methods directly from other scripts
    window.ElectricityPlanSelection = ElectricityPlanSelection;
  });
})(jQuery);
