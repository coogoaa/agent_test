## 2. 第一年节省

### 2.1 计算公式

```javascript
// 文件: result_debug.js

var firstYearSavings = calculateYearlyValues(values, 0, options).savings;
```

### 2.2 组成部分

```javascript
firstYearSavings = selfConsumptionSavings + FiTIncome + batterySavings

// 其中:
// selfConsumptionSavings = 自用电量 × 电价
// FiTIncome = 出口电量 × FiT
// batterySavings = 电池额外节省 (如有)
```

### 2.3 示例计算

```

电价: 27分

自用节省: 4919 × 0.27 = $1,328.13
FiT收入: 4530 × 0.07 = $317.10

```

## 3. 简单回收期

### 3.1 计算公式

```javascript
// 文件: shared_calc.js - detailedPayback()

function calculatePayback(totalCost, yearlyResults) {
    var cumulative = 0;
    
    for (var year = 0; year < yearlyResults.length; year++) {
        cumulative += yearlyResults[year].savings;
        
        if (cumulative >= totalCost) {
            // 计算精确回收月份
            var previousCumulative = cumulative - yearlyResults[year].savings;
            var remaining = totalCost - previousCumulative;
            var monthsInYear = remaining / yearlyResults[year].savings * 12;
            
            return year + monthsInYear / 12;
        }
    }
    
    return -1; // 未能回收
}
```

### 3.2 详细回收期计算

```javascript
// 文件: shared_calc.js

sCalc.detailedPayback = function(values, options) {
    var cumulativeSavings = 0;
    var paybackYear = -1;
    var paybackMonth = 0;
    
    for (var year = 0; year < 30; year++) {
        var result = sCalc.solarCalc.calculateYearlyValues(values, year, options);
        var previousCumulative = cumulativeSavings;
        cumulativeSavings += result.savings;
        
        if (paybackYear == -1 && cumulativeSavings >= values.total_cost) {
            paybackYear = year;
            
            // 计算精确月份
            var remaining = values.total_cost - previousCumulative;
            var monthlySavings = result.savings / 12;
            paybackMonth = Math.ceil(remaining / monthlySavings);
            
            break;
        }
    }
    
    return {
        years: paybackYear,
        months: paybackMonth,
        formatted: paybackYear + ' years ' + paybackMonth + ' months'
    };
}
```

### 3.3 示例计算

```
系统成本: $15,120
第1年节省: $1,645
第2年节省: $1,695
...
第9年节省: $2,085
第10年节省: $2,148

累计节省:
Year 1: $1,645
Year 2: $3,340
Year 3: $5,086
Year 4: $6,886
Year 5: $8,742
Year 6: $10,657
Year 7: $12,633
Year 8: $14,673
Year 9: $16,758  ← 超过 $15,120

精确计算:
Year 8 累计: $14,673
剩余: $15,120 - $14,673 = $447
Year 9 月均: $2,085 / 12 = $173.75
月份: $447 / $173.75 = 2.57 ≈ 3个月

回收期: 8年3个月
```

### 3.4 显示格式

```javascript

    $('#payback-period').text(payback.years + ' years ' + payback.months + ' months');
}
```

```
太阳能前:
- 用电量: 1110 kWh
- 电价: 27分
- 日供电费: 122.30
- 电费: 1110 × 0.27 + 122.30 = $422.00

- 自用量: 1110 kWh (全部自用)

```

## 7. 电池专用回收期

### 7.1 计算逻辑

```javascript
// 文件: result_debug.js

// 计算纯电池回收期 (不含太阳能)
function calculateBatteryOnlyPayback(values, options) {
    var batteryCost = values.battery_cost;
    var cumulativeBatterySavings = 0;
    
    for (var year = 0; year < 30; year++) {
        // 只计算电池带来的额外节省
        var withBattery = calculateYearlyValues(values, year, { ...options, battery: true });
        var withoutBattery = calculateYearlyValues(values, year, { ...options, battery: false });
        
        var batterySavings = withBattery.savings - withoutBattery.savings;
        cumulativeBatterySavings += batterySavings;
        
        if (cumulativeBatterySavings >= batteryCost) {
            return year + 1;
        }
    }
    
    return -1;
}
```

### 7.2 示例

```
电池成本: $10,500
年电池节省: ~$500

电池回收期: $10,500 / $500 ≈ 21年
```

## 8. 结果页面更新

### 8.1 refreshCalc 函数

```javascript
// 文件: result_debug.js

function refreshCalc() {
    // 1. 获取当前参数
    var selfC = $('#self-consumption-slider').slider('value') / 100;
    var inflation = $('#inflation-slider').slider('value') / 100;
    
    // 2. 计算20年结果
    var yearlyResults = [];
    for (var year = 0; year < 20; year++) {
        yearlyResults.push(calculateYearlyValues(variables, year, {
            selfC: selfC,
            inflation: inflation
        }));
    }
    
    // 3. 更新显示
    updateFirstYearSavings(yearlyResults[0].savings);
    updatePaybackPeriod(yearlyResults);
    updateCumulativeSavings(yearlyResults);
    updateSeasonalBills(yearlyResults[0]);
    updateCharts(yearlyResults);
}
```

### 8.2 滑块事件绑定

```javascript
// 自用率滑块
$('#self-consumption-slider').slider({
    min: 0,
    max: 100,
    value: variables.selfC,
    slide: function(event, ui) {
        refreshCalc();
    }
});

// 通胀率滑块
$('#inflation-slider').slider({
    min: 0,
    max: 10,
    step: 0.5,
    value: variables.inflation,
    slide: function(event, ui) {
        refreshCalc();
    }
});
```
